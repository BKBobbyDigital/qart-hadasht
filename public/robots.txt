# Strategy: allow traditional search and AI live-search/answer bots
# that cite sources; block AI training scrapers that don't.
# Cooperative protocol — bad-faith scrapers ignore this; the major AI
# companies have made public commitments to honor it for the bot
# identifiers below.

# ─── Default: allow everything not explicitly blocked ────────────────
User-agent: *
Allow: /

# ─── AI live-search / answer bots — ALLOW ────────────────────────────
# These fetch in real time when a user asks an AI a question, and
# cite the source in the response. This is the AI equivalent of
# search traffic and exactly the discoverability we want.

# OpenAI — live browsing in ChatGPT
User-agent: ChatGPT-User
Allow: /

# OpenAI — SearchGPT product
User-agent: OAI-SearchBot
Allow: /

# Anthropic — Claude live-fetch when users ask Claude things
User-agent: Claude-Web
Allow: /

User-agent: claude-web
Allow: /

# Perplexity — search/answer engine

User-agent: Perplexity-User
Allow: /

# You.com — answer engine
User-agent: YouBot
Allow: /

# Apple — Siri / Spotlight Suggestions
User-agent: Applebot
Allow: /

# out of training scrapes is a way of enforcing that requirement
# proactively.
